# UI demo elf (fake PAN) for elfloader3-linux: see run.sh
cmake_minimum_required(VERSION 3.28.2)
include(../../../sdk/config.cmake)
project(bt-pan-ui-demo C)
link_libraries(crt crt_helper gcc)
add_definitions(-DSWILIB_MODERN -D__NO_LIBC)
add_executable(ui_demo ui_demo.c ../../src/freestanding.c)
target_include_directories(ui_demo PRIVATE ../../src "${CMAKE_CURRENT_BINARY_DIR}")
# UI_PLATFORM=NSG (S75: 132x176, no ELKA) for elfloader3-nsg; default ELKA (EL71 240x320)
if(NOT UI_PLATFORM)
	set(UI_PLATFORM ELKA)
endif()
target_sdk_setup(ui_demo ${UI_PLATFORM})
target_compile_options(ui_demo BEFORE PUBLIC -std=c11 -Wall -Werror -Wno-error=main -Wno-unused-function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build_id.h" "#define BT_PAN_BUILD_TAG \"demo\"\n")
